#Requires AutoHotkey v2.0
#SingleInstance Force
Persistent

; Принудительно устанавливаем кодировку UTF-8 для логов
FileEncoding "UTF-8"

; --- Настройки путей и прокси ---
SingBoxExe   := "C:\sing-box\sing-box.exe"
ConfigDir    := "C:\sing-box"
ConfigFile   := "config.json"
LogFile      := A_ScriptDir "\ahk_debug.log"

; Адрес и порт вашего inbound типа "mixed" из config.json
ProxyAddress := "127.0.0.1:2080"

; --- Ваши иконки ---
IconOn  := A_ScriptDir "\icon_on.ico"
IconOff := A_ScriptDir "\icon_off.ico"

; Гарантированный сброс прокси и остановка процесса при закрытии скрипта
OnExit(CleanupOnExit)

; --- Получение версии ---
SBVersion := "sing-box"
try {
    RunWait(A_ComSpec ' /c ""' SingBoxExe '" version > "' A_Temp '\sb_ver.txt""',, "Hide")
    versionText := FileRead(A_Temp "\sb_ver.txt")
    SBVersion := StrSplit(versionText, "`n")[1]
}

Tray := A_TrayMenu
Global LastState := -1 

SetTimer(UpdateMenu, 1000)
UpdateMenu()

; --- Логика обновления меню ---
UpdateMenu() {
    Global LastState
    currentState := ProcessExist("sing-box.exe") ? 1 : 0
    
    ; Если sing-box внезапно упал, автоматически отключаем системный прокси
    if (!currentState && LastState == 1) {
        SetSystemProxy(false)
    }
    
    if (currentState == LastState)
        return
        
    LastState := currentState
    Tray.Delete()
    
    if (currentState) {
        if FileExist(IconOn)
            TraySetIcon(IconOn)
            
        Tray.Add("Статус: Подключен (Mixed)", (*) => "")
        if FileExist(IconOn)
            Tray.SetIcon("Статус: Подключен (Mixed)", IconOn)
        
        Tray.Add("Отключить VPN", ToggleVPN)
    } else {
        if FileExist(IconOff)
            TraySetIcon(IconOff)
            
        Tray.Add("Статус: Отключен", (*) => "")
        if FileExist(IconOff)
            Tray.SetIcon("Статус: Отключен", IconOff)
        
        Tray.Add("Включить VPN", ToggleVPN)
    }
    
    Tray.Add() ; Разделитель
    
    Tray.Add(SBVersion, (*) => "")
    Tray.Disable(SBVersion)
    
    Tray.Add("Выход", QuitApp)
}

; --- Логика переключения ---
ToggleVPN(*) {
    if ProcessExist("sing-box.exe") {
        FileAppend(A_Now " - Останавливаем процесс...`n", LogFile)
        ProcessClose("sing-box.exe")
        SetSystemProxy(false)
    } else {
        FileAppend(A_Now " - Пытаемся запустить...`n", LogFile)
        try {
            Run('"' SingBoxExe '" run -c "' ConfigFile '"', ConfigDir, "Hide")
            FileAppend(A_Now " - Команда Run отправлена успешно`n", LogFile)
            SetSystemProxy(true)
        } catch as err {
            FileAppend(A_Now " - ОШИБКА AHK: " err.Message "`n", LogFile)
        }
    }
    UpdateMenu()
}

; --- Управление системным прокси (без прав админа) ---
SetSystemProxy(enable := true) {
    regKey := "HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Internet Settings"
    if (enable) {
        RegWrite(1, "REG_DWORD", regKey, "ProxyEnable")
        RegWrite(ProxyAddress, "REG_SZ", regKey, "ProxyServer")
    } else {
        RegWrite(0, "REG_DWORD", regKey, "ProxyEnable")
    }
    
    ; Оповещение WinINet об изменении настроек (39 = SETTINGS_CHANGED, 37 = REFRESH)
    DllCall("wininet\InternetSetOptionW", "ptr", 0, "uint", 39, "ptr", 0, "uint", 0)
    DllCall("wininet\InternetSetOptionW", "ptr", 0, "uint", 37, "ptr", 0, "uint", 0)
}

CleanupOnExit(*) {
    SetSystemProxy(false)
    if ProcessExist("sing-box.exe") {
        ProcessClose("sing-box.exe")
    }
}

QuitApp(*) {
    ExitApp()
}
